cmake_minimum_required(VERSION 3.16)
project(rf_native_transport LANGUAGES CXX)

add_library(rf_native_transport STATIC src/rf_native_transport.cpp)
target_include_directories(rf_native_transport PUBLIC include)
target_compile_features(rf_native_transport PUBLIC cxx_std_17)
target_compile_options(rf_native_transport PRIVATE -Wall -Wextra -Wpedantic -Werror)

option(RNT_BUILD_TESTS "Build injectable-IO native transport tests" ON)
if(RNT_BUILD_TESTS)
  enable_testing()
  add_executable(rf_native_transport_tests tests/rf_native_transport_tests.cpp)
  target_link_libraries(rf_native_transport_tests PRIVATE rf_native_transport)
  target_compile_features(rf_native_transport_tests PRIVATE cxx_std_17)
  target_compile_options(rf_native_transport_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  find_package(Threads REQUIRED)
  target_link_libraries(rf_native_transport_tests PRIVATE Threads::Threads)
  add_test(NAME rf_native_transport_tests COMMAND rf_native_transport_tests)
endif()
