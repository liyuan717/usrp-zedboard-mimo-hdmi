    src/protocol.cpp
    src/rgb_expand.cpp
    src/presenter.cpp
    src/null_backend.cpp
    src/drm_backend.cpp)
set(HDMI_PRESENTER_CRC32_HEADER_DIR
    "${CMAKE_CURRENT_SOURCE_DIR}/../video_codec_helper/diagnostics"
    CACHE PATH "Directory containing canonical slice-by-8 CRC headers")
if(NOT EXISTS "${HDMI_PRESENTER_CRC32_HEADER_DIR}/crc32_slice8_candidate.h" OR
   NOT EXISTS "${HDMI_PRESENTER_CRC32_HEADER_DIR}/crc32_slice8_unroll_candidate.h")
  message(FATAL_ERROR "missing baseline or canonical unroll4 CRC header; do not fall back to bitwise CRC")
endif()
target_include_directories(hdmi_presenter_core PUBLIC include PRIVATE "${HDMI_PRESENTER_CRC32_HEADER_DIR}")
target_link_libraries(hdmi_presenter_core PUBLIC pthread)

add_executable(hdmi-presenter src/main.cpp src/receiver.cpp src/app_runner.cpp)
target_link_libraries(hdmi-presenter PRIVATE hdmi_presenter_core)

# The target ABI check is built only when pointed at the exact kernel UAPI tree
# used by the ARM image.  It compares every user-space structure and ioctl
# number used by the presenter, rather than relying on one aggregate size.
set(HDMI_PRESENTER_KERNEL_UAPI_DIR "" CACHE PATH "Kernel include/uapi directory for ABI parity")
set(HDMI_PRESENTER_ADI_UAPI_DIR "" CACHE PATH "Paired ADI HDMI driver's include/uapi directory for ABI parity")
set(HDMI_PRESENTER_KERNEL_SOURCE_DIR "" CACHE PATH "Kernel source root for UAPI internal include closure")
if(HDMI_PRESENTER_KERNEL_UAPI_DIR)
  if(NOT EXISTS "${HDMI_PRESENTER_KERNEL_UAPI_DIR}/drm/drm.h")
    message(FATAL_ERROR "HDMI_PRESENTER_KERNEL_UAPI_DIR must contain drm/drm.h")
  endif()
  if(NOT HDMI_PRESENTER_ADI_UAPI_DIR OR NOT EXISTS "${HDMI_PRESENTER_ADI_UAPI_DIR}/drm/adi_hdmi_present.h")
    message(FATAL_ERROR "HDMI_PRESENTER_ADI_UAPI_DIR must contain drm/adi_hdmi_present.h")
  endif()
  add_executable(uapi_parity_test tests/uapi_parity_test.cpp)
  target_include_directories(uapi_parity_test PRIVATE include "${HDMI_PRESENTER_ADI_UAPI_DIR}" "${HDMI_PRESENTER_KERNEL_UAPI_DIR}" "${HDMI_PRESENTER_KERNEL_UAPI_DIR}/drm")
  # Kernel source UAPI headers use GNU named variadic macros in declarations;
  # suppress only that imported-header diagnostic, retaining Werror for ours.
  target_compile_options(uapi_parity_test PRIVATE -Wno-variadic-macros)
  if(HDMI_PRESENTER_KERNEL_SOURCE_DIR)
    target_include_directories(uapi_parity_test PRIVATE
      "${HDMI_PRESENTER_KERNEL_SOURCE_DIR}/include"
      "${HDMI_PRESENTER_KERNEL_SOURCE_DIR}/arch/arm/include/uapi"
      "${HDMI_PRESENTER_KERNEL_SOURCE_DIR}/arch/arm/include")
    target_compile_definitions(uapi_parity_test PRIVATE __EXPORTED_HEADERS__=1)
  endif()
  target_compile_options(uapi_parity_test PRIVATE -UNDEBUG)
  add_test(NAME uapi_parity_test COMMAND uapi_parity_test)
endif()
